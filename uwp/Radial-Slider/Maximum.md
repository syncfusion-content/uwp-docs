# Maximum in UWP Radial Slider (SfRadialSlider)

Gets or sets the maximum possible value.

{% tabs %}

{% highlight xaml %}

<syncfusion:SfRadialSlider x:Name="radialSlider"

            Minimum="0" 

            Maximum="100"  />

{% endhighlight %}

{% endtabs %}

{% tabs %}

{% highlight C# %}

radialSlider.Minimum = 0;

radialSlider.Maximum = 100;

{% endhighlight %}

{% highlight VB %}

radialSlider.Minimum = 0

radialSlider.Maximum = 100

{% endhighlight %}

{% endtabs %}
